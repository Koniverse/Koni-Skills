```ts
/**
 * Find the first heading whose rendered text **starts with** `prefix`
 * (case-sensitive, leading/trailing whitespace tolerant), returning the
 * same SectionMatch shape as findSection. Useful for numbered section
 * lookups where downstream text varies (e.g. "## 8. Functional Requirements (FR)"
 * vs "## 8. Functional requirements").
 *
 * @param doc  parsed Doc
 * @param prefix  e.g. '## 8.' (depth marker + number-and-dot).
 *                The leading '#'+spaces count is inferred from prefix's leading '#' chars.
 *                Heading text is matched against the suffix after that.
 */
export function findSectionStartingWith(doc: Doc, prefix: string): SectionMatch | null {
  // Parse the heading depth + text portion of the prefix.
  const m = prefix.match(/^(#+)\s+(.*)$/);
  if (!m) return null;
  const depth = m[1]!.length;
  const textPrefix = m[2]!.trim();
  for (let i = 0; i < doc.ast.children.length; i++) {
    const node = doc.ast.children[i];
    if (node?.type !== 'heading') continue;
    if (node.depth !== depth) continue;
    const text = nodeToString(node).trim();
    if (!text.startsWith(textPrefix)) continue;
    const startDepth = node.depth;
    let end = doc.ast.children.length;
    for (let j = i + 1; j < doc.ast.children.length; j++) {
      const next = doc.ast.children[j];
      if (next?.type === 'heading' && next.depth <= startDepth) {
        end = j;
        break;
      }
    }
    return {
      heading: node,
      headingIndex: i,
      body: doc.ast.children.slice(i + 1, end) as Content[],
      endIndex: end,
    };
  }
  return null;
}
```

- [ ] **Step 4: Update `parseDoc` to detect quoted keys**

Open `packages/koni-docs/src/lib/doc.ts`. Replace the file with:

```ts
import { readFileSync, writeFileSync } from 'node:fs';
import matter from 'gray-matter';
import { parseMarkdown, stringifyMarkdown } from './markdown/ast.ts';
import type { Doc } from './types.ts';

// Match a single-line YAML scalar with a quoted value at the top level of frontmatter.
// Example: `version_shipped: "0.6.0"` or `title: 'Foo bar'`.
const QUOTED_VALUE_RE = /^(\w[\w-]*):\s*(["']).*\2\s*$/;

function detectQuotedKeys(raw: string): Set<string> {
  const out = new Set<string>();
  // Frontmatter is between two `---` lines at file start.
  const m = raw.match(/^---\n([\s\S]*?)\n---/);
  if (!m) return out;
  const fmBody = m[1] ?? '';
  for (const line of fmBody.split('\n')) {
    const km = line.match(QUOTED_VALUE_RE);
    if (km) out.add(km[1]!);
  }
  return out;
}

export function readDoc(path: string): Doc {
  const raw = readFileSync(path, 'utf-8');
  return parseDoc(raw, path);
}

export function parseDoc(raw: string, path: string): Doc {
  const parsed = matter(raw);
  const body = parsed.content;
  const frontmatterQuoting = detectQuotedKeys(raw);
  return {
    path,
    frontmatter: { ...parsed.data },
    body,
    ast: parseMarkdown(body),
    raw,
    frontmatterQuoting,
  };
}

export function serializeDoc(doc: Doc): string {
  // Re-stringify AST → body to capture any AST edits.
  const newBody = stringifyMarkdown(doc.ast);
  // gray-matter.stringify handles the frontmatter block.
  const out = matter.stringify(newBody, doc.frontmatter);
  if (!doc.frontmatterQuoting || doc.frontmatterQuoting.size === 0) return out;
  return reapplyQuoting(out, doc.frontmatterQuoting);
}

/**
 * Post-process gray-matter's output: for each key in `quotedKeys`, wrap its
 * value in double-quotes if gray-matter emitted it bare. Only touches the
 * frontmatter block (between the first two `---` lines).
 */
function reapplyQuoting(serialized: string, quotedKeys: Set<string>): string {
  const m = serialized.match(/^(---\n)([\s\S]*?)(\n---\n?)([\s\S]*)$/);
  if (!m) return serialized;
  const [, open, fmBody, close, rest] = m;
  const patched = (fmBody ?? '').split('\n').map(line => {
    const km = line.match(/^(\w[\w-]*):\s*(.*)$/);
    if (!km) return line;
    const key = km[1]!;
    const value = km[2] ?? '';
    if (!quotedKeys.has(key)) return line;
    // Skip if already quoted (single or double).
    if (/^(['"]).*\1$/.test(value)) return line;
    // Skip empty or numeric block values we shouldn't touch.
    if (value === '' || value === 'null' || value === '~') return line;
    return `${key}: "${value.replace(/"/g, '\\"')}"`;
  }).join('\n');
  return `${open}${patched}${close}${rest}`;
}

export function writeDoc(path: string, doc: Doc): void {
  writeFileSync(path, serializeDoc(doc), 'utf-8');
}

export function updateFrontmatter(doc: Doc, partial: Record<string, unknown>): Doc {
  return { ...doc, frontmatter: { ...doc.frontmatter, ...partial } };
}
```

```ts
export interface FrRefMissing {
  /** Story id (e.g. "US-4.1"). */
  id: string;
  /** Story file path. */
  source: string;
  /** FR refs that the story claims but PRD §8 table does not contain. */
  missingFr: string[];
}

/**
 * For each story with `prd_ref:` frontmatter, verify each FR-N is present as a
 * row in PRD §8's table. Returns a list of stories with at least one missing
 * FR ref. Empty list = all FR refs resolve.
 */
export function validateFrRefs(corpus: Corpus): FrRefMissing[] {
  const prdEntry = corpus.singletons.prd;
  if (!prdEntry) return [];
  // Parse the PRD to AST so we can locate §8's table cleanly.
  // We re-use parseDoc + findSectionStartingWith + parseTable from the lib.
  // Avoid circular import: do the lookup inline using gray-matter parts + readDoc.
  const { readDoc } = require('./doc.ts') as typeof import('./doc.ts');
  const { findSectionStartingWith } = require('./markdown/sections.ts') as typeof import('./markdown/sections.ts');
  const { findTable, parseTable } = require('./markdown/tables.ts') as typeof import('./markdown/tables.ts');
  const prdDoc = readDoc(prdEntry.path);
  const section8 = findSectionStartingWith(prdDoc, '## 8.');
  if (!section8) return getStories(corpus).map(s => storyFrRefsAll(s)).filter(x => x !== null) as FrRefMissing[];
  // Find the first table inside §8 body.
  const tableNode = section8.body.find(n => n.type === 'table');
  if (!tableNode || tableNode.type !== 'table') return [];
  const parsed = parseTable(tableNode);
  const idCol = parsed.headers.indexOf('ID');
  const validIds = new Set<string>();
  if (idCol >= 0) for (const row of parsed.rows) validIds.add(row[idCol] ?? '');

  const out: FrRefMissing[] = [];
  for (const s of getStories(corpus)) {
    const refs = refsFromStory(s).filter(r => r.kind === 'prd_ref').map(r => r.ref);
    const missing = refs.filter(r => /^FR-/.test(r) && !validIds.has(r));
    if (missing.length > 0) {
      out.push({
        id: String(s.frontmatter.id ?? ''),
        source: s.path,
        missingFr: missing,
      });
    }
  }
  return out;
}

// Used only by the early-return branch when §8 cannot be located.
function storyFrRefsAll(s: MatterEntry): FrRefMissing | null {
  const refs = refsFromStory(s).filter(r => r.kind === 'prd_ref').map(r => r.ref);
  const missing = refs.filter(r => /^FR-/.test(r));
  if (missing.length === 0) return null;
  return {
    id: String(s.frontmatter.id ?? ''),
    source: s.path,
    missingFr: missing,
  };
}
```

```ts
export function registerValidate(program: Command): void {

    .command('validate')
    .description('Validate the L3 ID graph (epic/sprint/PRD references) — exits non-zero on error')
    .option('--json', 'machine-readable output', false)
    .option('--include-warnings', 'include FR-refs that have no PRD §8 row (default: warnings = errors)', false)
    .action(function (this: Command, cmdOpts: ValidateFlags) {

      const corpus = loadCorpus(opts.docsPath);
      const refErrors = validateRefs(corpus);
      const frMissing = validateFrRefs(corpus);

      const errorCount = refErrors.length + frMissing.length;

      if (cmdOpts.json || opts.json) {
        console.log(JSON.stringify({
          ok: errorCount === 0,
          refErrors,
          frMissing,
          summary: {
            ref: refErrors.length,
            fr: frMissing.length,
          },
        }, null, 2));
      } else {
        console.log(`koni-docs validate — ${opts.docsPath}`);
        console.log('');
        if (refErrors.length === 0 && frMissing.length === 0) {
          console.log('  ✓ all references resolve');
        } else {
          if (refErrors.length > 0) {
            console.log(`  ✗ ${refErrors.length} reference error(s):`);
            for (const e of refErrors) {
              console.log(`    - ${e.source}: ${e.kind}="${e.ref}" (${e.error})`);
            }
          }
          if (frMissing.length > 0) {
            console.log(`  ✗ ${frMissing.length} FR-ref miss(es):`);
            for (const m of frMissing) {
              console.log(`    - ${m.id} (${m.source}): missing ${m.missingFr.join(', ')} in PRD §8`);
            }
          }
        }
      }
      process.exit(errorCount === 0 ? 0 : 1);
    });
```

```ts
import { existsSync, readFileSync } from 'node:fs';
import { join } from 'node:path';
import { pathToFileURL } from 'node:url';
import { z } from 'zod';

const ConfigSchema = z.object({
  title: z.string().optional(),
  folderOrder: z.array(z.string()).optional(),
  topLevelOrder: z.array(z.string()).optional(),
});

export type KoniDocsViewerConfig = z.infer<typeof ConfigSchema>;

/**
 * Look for `koni-docs.config.json` then `koni-docs.config.mjs` at the docs-tree
 * root. Validate against the schema. Returns `{}` if neither exists. Throws
 * with a clear message on schema error.
 */
export function loadViewerConfig(docsDir: string): KoniDocsViewerConfig {
  const jsonPath = join(docsDir, 'koni-docs.config.json');
  const mjsPath = join(docsDir, 'koni-docs.config.mjs');

  let raw: unknown = null;
  let source = '';
  if (existsSync(jsonPath)) {
    source = jsonPath;
    try {
      raw = JSON.parse(readFileSync(jsonPath, 'utf-8'));
    } catch (e) {
      throw new Error(`koni-docs.config.json: invalid JSON — ${(e as Error).message}`);
    }
  } else if (existsSync(mjsPath)) {
    source = mjsPath;
    // Synchronous dynamic-import workaround: use require via createRequire so the
    // function stays sync. .mjs files are loaded via dynamic import which is async,
    // so for .mjs we fall back to spawn an inline node -e snippet. Simpler: switch
    // to async API — but Astro corpus is async-friendly. We keep this sync for
    // consistency by using a top-level `require`-equivalent: read the file and parse
    // its default export via a regex on `export default { ... }`. For full ESM
    // support, callers should prefer the JSON form.
    //
    // Pragmatic compromise: support both by spawning child node to evaluate the
    // .mjs and capture its default-export as JSON.
    const childOut = childResolveMjs(mjsPath);
    raw = childOut;
  } else {
    return {};
  }

  const parsed = ConfigSchema.safeParse(raw);
  if (!parsed.success) {
    const issues = parsed.error.issues.map(i => `  ${i.path.join('.')}: ${i.message}`).join('\n');
    throw new Error(`${source}: schema validation failed:\n${issues}`);
  }
  return parsed.data;
}

function childResolveMjs(mjsPath: string): unknown {
  // Use spawnSync to load the .mjs and dump its default export as JSON.
  // Avoids async wrapping at the caller and works with any node 20+ ESM module.
  const { spawnSync } = require('node:child_process') as typeof import('node:child_process');
  const url = pathToFileURL(mjsPath).href;
  const script = `import(${JSON.stringify(url)}).then(m => process.stdout.write(JSON.stringify(m.default ?? m))).catch(e => { process.stderr.write(String(e)); process.exit(1); });`;
  const r = spawnSync(process.execPath, ['--input-type=module', '-e', script], { encoding: 'utf-8' });
  if (r.status !== 0) {
    throw new Error(`${mjsPath}: failed to load — ${r.stderr.trim()}`);
  }
  try {
    return JSON.parse(r.stdout);
  } catch (e) {
    throw new Error(`${mjsPath}: default export is not JSON-serializable — ${(e as Error).message}`);
  }
}
```

```ts
import { EventEmitter } from 'node:events';
import chokidar from 'chokidar';

interface ReloadEvent {
  path: string;
  kind: 'change' | 'add' | 'unlink';
}

class ReloadBus extends EventEmitter {
  private watcher: chokidar.FSWatcher | null = null;
  private timer: NodeJS.Timeout | null = null;
  private lastEvent: ReloadEvent | null = null;
  private subscribers = 0;

  start(docsDir: string): void {
    if (this.watcher) return;
    this.watcher = chokidar.watch(docsDir, {
      ignored: /(\/node_modules\/|\/\.git\/|\/\.astro\/)/,
      ignoreInitial: true,
    });
    const fire = (kind: ReloadEvent['kind']) => (filePath: string) => {
      if (!filePath.endsWith('.md')) return;
      this.lastEvent = { path: filePath, kind };
      if (this.timer) clearTimeout(this.timer);
      this.timer = setTimeout(() => {
        if (this.lastEvent) this.emit('reload', this.lastEvent);
        this.lastEvent = null;
        this.timer = null;
      }, 300);
    };
    this.watcher.on('change', fire('change'));
    this.watcher.on('add', fire('add'));
    this.watcher.on('unlink', fire('unlink'));
  }

  /** Subscribe a listener and return a disposer that decrements the count + removes it. */
  subscribe(listener: (e: ReloadEvent) => void): () => void {
    this.on('reload', listener);
    this.subscribers++;
    return () => {
      this.off('reload', listener);
      this.subscribers--;
    };
  }

  subscriberCount(): number {
    return this.subscribers;
  }
}

let busSingleton: ReloadBus | null = null;

export function getReloadBus(docsDir: string): ReloadBus {
  if (!busSingleton) {
    busSingleton = new ReloadBus();
    if (process.env.KONI_DOCS_WATCH === '1') {
      busSingleton.start(docsDir);
    }
  }
  return busSingleton;
}

export type { ReloadEvent };
```

- [ ] **Step 3: Create the SSE endpoint at `src/viewer/pages/__koni-docs/reload.ts`**

```ts
import type { APIRoute } from 'astro';
import { getReloadBus, type ReloadEvent } from '../../lib/reload-bus.ts';
import { getDocsDir } from '../../lib/corpus.ts';

export const GET: APIRoute = ({ request }) => {
  const bus = getReloadBus(getDocsDir());
  const encoder = new TextEncoder();

  let unsubscribe: (() => void) | null = null;

  const stream = new ReadableStream({
    start(controller) {
      // Initial comment to flush headers.
      controller.enqueue(encoder.encode(': connected\n\n'));

      const listener = (e: ReloadEvent) => {
        controller.enqueue(encoder.encode(`data: ${JSON.stringify(e)}\n\n`));
      };
      unsubscribe = bus.subscribe(listener);

      // Keepalive comment every 25s to keep proxies happy.
      const keepalive = setInterval(() => {
        try { controller.enqueue(encoder.encode(': keepalive\n\n')); } catch { clearInterval(keepalive); }
      }, 25_000);

      // Client disconnect handler — runs when the browser closes EventSource.
      request.signal.addEventListener('abort', () => {
        clearInterval(keepalive);
        if (unsubscribe) unsubscribe();
        try { controller.close(); } catch {}
      });
    },
    cancel() {
      if (unsubscribe) unsubscribe();
    },
  });

  return new Response(stream, {
    headers: {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache, no-transform',
      'Connection': 'keep-alive',
      'X-Accel-Buffering': 'no',
    },
  });
};
```
